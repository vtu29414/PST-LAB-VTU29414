Question:
You are given a string s of even length. 
Split this string into two halves of equal lengths, and let a be the first half and b be the second half. 
Two strings are alike if they have the same number of vowels ('a', 'e', 'i', 'o', 'u', 'A', 'E', 'I', 'O', 'U'). 
Notice that s contains uppercase and lowercase letters. Return true if a and b are alike. Otherwise, return false. 
Example 1: Input: s = "book" 
Output: true
Explanation: a = "bo" and b = "ok". a has 1 vowel and b has 1 vowel. 
Therefore, they are alike.

Program:
import java.util.Scanner;

public class Main {

    public static boolean halvesAreAlike(String s) {
        String vowels = "aeiouAEIOU";
        int firstHalfCount = 0;
        int secondHalfCount = 0;

        int mid = s.length() / 2;

        // Count vowels in first half
        for (int i = 0; i < mid; i++) {
            if (vowels.indexOf(s.charAt(i)) != -1) {
                firstHalfCount++;
            }
        }

        // Count vowels in second half
        for (int i = mid; i < s.length(); i++) {
            if (vowels.indexOf(s.charAt(i)) != -1) {
                secondHalfCount++;
            }
        }

        return firstHalfCount == secondHalfCount;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        String s = sc.nextLine();

        System.out.println(halvesAreAlike(s));

        sc.close();
    }
}
